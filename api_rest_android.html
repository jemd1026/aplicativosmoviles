📚 PARTE I: FUNDAMENTOS TEÓRICOS

¿Qué vamos a construir?

Desarrollaremos una aplicación Android completa que consume datos de un servidor web mediante una API REST. La app mostrará un listado de productos obtenidos desde una base de datos MySQL a través de un backend PHP.

• RecyclerView para mostrar lista de productos

🔑 Conceptos Clave

📡 API REST

Representational State Transfer

  • Interfaz para comunicación entre sistemas
  • Usa protocolo HTTP/HTTPS
  • Intercambio de datos en JSON
  • Sin estado (stateless)
  • Recursos identificados por URLs

🔄 Retrofit

Cliente HTTP para Android

  • Librería de Square
  • Convierte API HTTP en interfaz Java/Kotlin
  • Serialización automática con Gson
  • Manejo de callbacks asíncronos
  • Interceptores y logging

♻️ RecyclerView

Vista de lista avanzada

  • Reemplazo mejorado de ListView
  • Reciclaje eficiente de vistas
  • Layouts flexibles (Linear, Grid, etc.)
  • Animaciones incorporadas
  • ViewHolder pattern obligatorio

📦 JSON

JavaScript Object Notation

  • Formato ligero de intercambio de datos
  • Fácil de leer y escribir
  • Soportado nativamente en PHP
  • Estructura de pares clave-valor
  • Arrays y objetos anidados

🗄️ MySQL + PDO

Base de datos y conexión

  • MySQL: Sistema de gestión de BD
  • PDO: PHP Data Objects
  • Prepared statements (seguridad)
  • Prevención de SQL injection
  • Transacciones y manejo de errores

🎯 Patrón MVC

Model-View-Controller

  • Model: Clases de datos (Producto)
  • View: Layouts XML
  • Controller: Activities y Adapters
  • Separación de responsabilidades
  • Código más mantenible

ProductoAdapter recibe la lista de productos

2️⃣ Backend PHP: API REST

📁 Estructura de carpetas:
htdocs/
└── ucv_aplic_mov/
    ├── config.php
    └── productos/

PHP - config.php (Conexión a la base de datos)
<?php
$host = "localhost";
$db = "aplic_movil";
$user = "root";
$pass = "";

try {
    $conexion = new PDO("mysql:host=$host;dbname=$db;charset=utf8", $user, $pass);
    $conexion->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
} catch (PDOException $e) {
    echo json_encode([
        "success" => false,
        "message" => "Error de conexión: " . $e->getMessage()
    ]);
    exit;
}
?>
PHP - productos/listar.php
<?php
// Incluir configuración
require_once '../config.php';

header('Content-Type: application/json; charset=utf-8');

try {
    // Consulta SQL para obtener todos los productos
    $query = "SELECT * FROM productos ORDER BY nombre ASC";
    $stmt = $conexion->prepare($query);
    $stmt->execute();

    // OJO: fetchAll(PDO::FETCH_ASSOC) elimina los índices numéricos
    $productos = $stmt->fetchAll(PDO::FETCH_ASSOC);

    // Devolver respuesta exitosa con los datos
    echo json_encode([
        "success" => true,
        "message" => "Productos obtenidos correctamente",
        "data" => $productos,
        "total" => count($productos)
    ], JSON_UNESCAPED_UNICODE);

} catch (PDOException $e) {
    // En caso de error
    echo json_encode([
        "success" => false,
        "message" => "Error al obtener productos: " . $e->getMessage()
    ]);
}

Adapter para RecyclerView

Kotlin - adapters/ProductoAdapter.kt
package com.example.apirestapp.adapters

import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.TextView
import androidx.recyclerview.widget.RecyclerView
import com.example.apirestapp.R
import com.example.apirestapp.models.Producto

class ProductoAdapter(
    private var items: List<Producto>
) : RecyclerView.Adapter<ProductoAdapter.ProductoViewHolder>() {

    class ProductoViewHolder(itemView: View) : RecyclerView.ViewHolder(itemView) {
        val tvNombre: TextView = itemView.findViewById(R.id.tvNombre)
        val tvCategoria: TextView = itemView.findViewById(R.id.tvCategoria)
        val tvPrecio: TextView = itemView.findViewById(R.id.tvPrecio)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): ProductoViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_producto, parent, false)
        return ProductoViewHolder(view)
    }

    override fun onBindViewHolder(holder: ProductoViewHolder, position: Int) {
        val producto = items[position]
        holder.tvNombre.text = producto.nombre
        holder.tvCategoria.text = "Categoría: ${producto.categoria}"
        holder.tvPrecio.text = "S/ ${producto.precio}"
    }

    override fun getItemCount(): Int = items.size

    fun actualizarDatos(nuevos: List<Producto>) {
        items = nuevos
        notifyDataSetChanged()
    }
}

Activity Principal

Kotlin - MainActivity.kt
package com.example.apirestapp

import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.widget.Button
import android.widget.Toast
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.example.apirestapp.adapters.ProductoAdapter
import com.example.apirestapp.api.RetrofitClient
import com.example.apirestapp.models.ProductosResponse
import retrofit2.Call
import retrofit2.Callback
import retrofit2.Response

class MainActivity : AppCompatActivity() {

    private lateinit var recyclerView: RecyclerView
    private lateinit var adapter: ProductoAdapter

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        // Configurar RecyclerView
        recyclerView = findViewById(R.id.recyclerProductos)
        recyclerView.layoutManager = LinearLayoutManager(this)

        // Inicializar adapter con lista vacía
        adapter = ProductoAdapter(emptyList())
        recyclerView.adapter = adapter

        // Cargar productos desde la API
        cargarProductos()
        ////////
        val btnRegistrar = findViewById<Button>(R.id.btnRegistrar)
        btnRegistrar.setOnClickListener {
            val intent = Intent(this, RegistrarActivity::class.java)
            startActivity(intent)
        }
    }

    private fun cargarProductos() {
        val call = RetrofitClient.instance.listarProductos()

        call.enqueue(object : Callback<ProductosResponse> {
            override fun onResponse(
                call: Call<ProductosResponse>,
                response: Response<ProductosResponse>
            ) {
                val body = response.body()
                if (response.isSuccessful && body != null && body.success) {
                    // Actualizar el adapter con los datos recibidos
                    adapter.actualizarDatos(body.data)
                } else {
                    Toast.makeText(
                        this@MainActivity,
                        "Error al cargar productos",
                        Toast.LENGTH_SHORT
                    ).show()
                }
            }

            override fun onFailure(call: Call<ProductosResponse>, t: Throwable) {
                Toast.makeText(
                    this@MainActivity,
                    "Error de conexión: ${t.message}",
                    Toast.LENGTH_SHORT
                ).show()
            }
        })
    }
}

📝 PARTE III: REGISTRO DE PRODUCTOS (INSERTAR)

1️⃣ Crear el endpoint registrar.php

Paso 1: Crea un archivo nuevo llamado registrar.php en la misma carpeta donde está listar.php.
Ruta: htdocs/ucv_aplic_mov/productos/registrar.php
PHP - productos/registrar.php
<?php
require_once '../config.php';

header('Content-Type: application/json; charset=utf-8');

// Obtener datos del POST
$nombre = $_POST['nombre'] ?? '';
$precio = $_POST['precio'] ?? '';
$categoria = $_POST['categoria'] ?? '';

if ($nombre && $precio && $categoria) {
    try {
        $query = "INSERT INTO productos (nombre, precio, categoria) VALUES (?, ?, ?)";
        $stmt = $conexion->prepare($query);
        $stmt->execute([$nombre, $precio, $categoria]);
        echo json_encode([
            "success" => true,
            "message" => "Producto registrado correctamente"
        ]);
    } catch (PDOException $e) {
        echo json_encode([
            "success" => false,
            "message" => "Error al registrar: " . $e->getMessage()
        ]);
    }
} else {
    echo json_encode([
        "success" => false,
        "message" => "Faltan datos obligatorios"
    ]);
}
?>

2️⃣ Crear RegistrarActivity y su layout

Paso 2: Crea un nuevo archivo RegistrarActivity.kt y su layout activity_registrar.xml:
Kotlin - RegistrarActivity.kt
package com.example.api_php

import android.os.Bundle
import android.widget.Button
import android.widget.EditText
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import retrofit2.Call
import retrofit2.Callback
import retrofit2.Response

class RegistrarActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_registrar)

        val etNombre = findViewById<EditText>(R.id.etNombre)
        val etPrecio = findViewById<EditText>(R.id.etPrecio)
        val etCategoria = findViewById<EditText>(R.id.etCategoria)
        val btnGuardar = findViewById<Button>(R.id.btnGuardar)

        btnGuardar.setOnClickListener {
            val nombre = etNombre.text.toString()
            val precio = etPrecio.text.toString()
            val categoria = etCategoria.text.toString()
            if (nombre.isNotBlank() && precio.isNotBlank() && categoria.isNotBlank()) {
                val call = RetrofitClient.instance.registrarProducto(nombre, precio, categoria)
                call.enqueue(object : Callback> {
                    override fun onResponse(call: Call>, response: Response>) {
                        val body = response.body()
                        if (body != null && body["success"] == true) {
                            Toast.makeText(this@RegistrarActivity, "Registro exitoso", Toast.LENGTH_SHORT).show()
                            finish()
                        } else {
                            Toast.makeText(this@RegistrarActivity, "Error: ${body?.get("message")}", Toast.LENGTH_SHORT).show()
                        }
                    }
                    override fun onFailure(call: Call>, t: Throwable) {
                        Toast.makeText(this@RegistrarActivity, "Error de red: ${t.message}", Toast.LENGTH_SHORT).show()
                    }
                })
            } else {
                Toast.makeText(this, "Completa todos los campos", Toast.LENGTH_SHORT).show()
            }
        }
    }
}
