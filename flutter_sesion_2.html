NUEVO lib/core/validators.dart 
// ======================================================
// core/validators.dart
// Funciones que revisan si lo que escribio el usuario
// es correcto. Las usaremos en el Login Y en el Registro,
// por eso viven aqui y no dentro de un formulario.
//
// Regla de TODO validator:
//   - devuelve null      -> el campo esta bien
//   - devuelve un String -> ese texto aparece en rojo
// ======================================================

class Validators {
  // 1) Campo obligatorio: no puede quedar vacio
  static String? obligatorio(String? valor) {
    // trim() quita los espacios: "   " tambien cuenta como vacio
    if (valor == null || valor.trim().isEmpty) {
      return 'Este campo es obligatorio';
    }
    return null; // null = sin error
  }

  // 2) Correo electronico
  static String? email(String? valor) {
    // Primero revisamos que no este vacio (reutilizamos la funcion 1)
    final error = obligatorio(valor);
    if (error != null) return error;

    // RegExp = un "molde" que el texto debe cumplir:
    //   letras/numeros  @  dominio  .  com / pe / edu ...
    final molde = RegExp(r'^[\w\.\-]+@([\w\-]+\.)+[a-zA-Z]{2,4}$');

    // valor! -> el "!" le dice a Dart: "tranquilo, aqui no es null"
    if (!molde.hasMatch(valor!.trim())) {
      return 'Correo electronico invalido';
    }
    return null;
  }

  // 3) Contraseña segura: 8 caracteres, una MAYUSCULA y un numero
  static String? password(String? valor) {
    final error = obligatorio(valor);
    if (error != null) return error;

    if (valor!.length < 8) {
      return 'Debe tener 8 caracteres como minimo';
    }
    // contains(RegExp(...)) pregunta: "¿hay al menos uno de estos?"
    if (!valor.contains(RegExp(r'[A-Z]'))) {
      return 'Debe tener al menos una letra MAYUSCULA';
    }
    if (!valor.contains(RegExp(r'[0-9]'))) {
      return 'Debe tener al menos un numero';
    }
    return null;
  }

  // 4) Repetir contraseña: debe ser igual a la primera
  //    Recibe 2 datos: lo escrito y la contraseña original
  static String? confirmar(String? valor, String original) {
    final error = obligatorio(valor);
    if (error != null) return error;

    if (valor != original) {
      return 'Las contraseñas no coinciden';
    }
    return null;
  }
}

// PARTE 1: el widget. Solo recibe datos del "padre".
class FormLogin extends StatefulWidget {
  const FormLogin({super.key, this.onTap});

  // Funcion que nos envia LoginScreen.
  // Se ejecuta cuando el usuario toca una caja de texto.
  final void Function()? onTap;

  @override
  State<FormLogin> createState() => _FormLoginState(); // conecta con su estado
}

// PARTE 2: el estado. Aqui viven las variables que cambian.
// El "_" al inicio significa: privado, solo se usa en este archivo.
class _FormLoginState extends State<FormLogin> {
  // Llave del formulario: nos deja preguntar "¿todo esta bien?"
  final formKey = GlobalKey<FormState>();

  // Controladores: guardan lo que el usuario escribe
  final emailController = TextEditingController();
  final passController = TextEditingController();

  // true = contraseña oculta (••••••)
  bool hidePassword = true;

  // dispose() se ejecuta al cerrar la pantalla.
  // Liberamos los controladores para no gastar memoria.

    emailController.dispose();
    passController.dispose();
    super.dispose();
  }

  // Se ejecuta al presionar "Ingresar"
  void ingresar() {
    // validate() ejecuta TODOS los validator del formulario
    if (formKey.currentState!.validate()) {
      // TODO (Bloque 2): aqui iremos a la pantalla Home
      mostrarMensaje('Bienvenido ${emailController.text}', Colors.green);
    } else {
      mostrarMensaje('Formulario invalido', Colors.red);
    }
  }

  // Muestra un mensaje flotante abajo (SnackBar)
  void mostrarMensaje(String texto, Color color) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(texto, style: const TextStyle(color: Colors.white)),
        backgroundColor: color,
        behavior: SnackBarBehavior.floating, // flota sobre la pantalla
        showCloseIcon: true,                 // muestra una X para cerrarlo
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Form(
      key: formKey, // conectamos la llave con este formulario
      child: Card(
        color: Colors.white,
        margin: EdgeInsets.zero, // sin margen: pegado a los bordes
        // Solo redondeamos las 2 esquinas de arriba
        shape: const RoundedRectangleBorder(
          borderRadius: BorderRadius.only(
            topLeft: Radius.circular(32),
            topRight: Radius.circular(32),
          ),
        ),
        child: Padding(
          padding: const EdgeInsets.all(16),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Text(
                'Ingreso',
                style: TextStyle(
                  fontWeight: FontWeight.bold,
                  fontSize: 24,
                  color: AppColors.texto,
                ),
              ),
              const SizedBox(height: 16),

              // ---------- CAJA 1: CORREO ----------
              TextFormField(
                controller: emailController,              // guarda lo escrito
                keyboardType: TextInputType.emailAddress, // teclado con @
                decoration: const InputDecoration(
                  border: OutlineInputBorder(),           // borde alrededor
                  isDense: true,                          // caja mas compacta
                  icon: Icon(Icons.email_outlined),       // icono a la izquierda
                  labelText: 'Correo electronico',
                ),
                // "widget." para leer datos de la PARTE 1 (el StatefulWidget)
                onTap: widget.onTap,
                // Cuando sale el teclado, sube la pantalla 400 px

import 'package:flutter/material.dart';
import 'package:app_flutter/core/theme.dart';
import 'package:app_flutter/core/validators.dart';
import 'package:app_flutter/presentation/widgets/app_button.dart';

// ======================================================
// Formulario de registro (4 cajas de texto).
// ======================================================

class FormRegister extends StatefulWidget {
  const FormRegister({super.key, this.onTap});

  final void Function()? onTap;

  @override
  State<FormRegister> createState() => _FormRegisterState();
}

class _FormRegisterState extends State<FormRegister> {
  final formKey = GlobalKey<FormState>();

  // Un controlador por cada caja
  final nombreController = TextEditingController();
  final emailController = TextEditingController();
  final passController = TextEditingController();
  final pass2Controller = TextEditingController();

  bool hidePassword = true; // controla las DOS cajas de contraseña

  @override
  void dispose() {
    nombreController.dispose();
    emailController.dispose();
    passController.dispose();
    pass2Controller.dispose();
    super.dispose();
  }

  void registrar() {
    // Si algun validator devuelve error, no hacemos nada
    if (!formKey.currentState!.validate()) return;

    // Todo bien: mostramos un mensaje...
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text('Cuenta creada para ${nombreController.text}'),
        backgroundColor: Colors.green,
        behavior: SnackBarBehavior.floating,
      ),
    );

    // ...y volvemos al Login (quitamos Registro de la pila)
    Navigator.pop(context);

// ======================================================
// data/models/producto.dart
// MODELO: describe COMO es un producto de la bodega.
// No dibuja nada en pantalla, solo guarda datos.
// ======================================================

class Producto {
  // Los "casilleros" de la ficha
  final String nombre;
  final String categoria;
  final double precio; // double = numero con decimales
  final int stock;     // int = numero entero (unidades)

  // Constructor con parametros con nombre: Producto(nombre: ..., precio: ...)
  const Producto({
    required this.nombre,
    required this.categoria,
    required this.precio,
    required this.stock,
  });

  // Getter = dato CALCULADO. Se usa como variable: producto.stockBajo
  // Devuelve true si quedan menos de 5 unidades.
  bool get stockBajo => stock < 5;
}

Paso 2 · La vista de la lista

ListView.builder funciona como un bucle for que dibuja: le decimos cuantos elementos hay (itemCount) y como dibujar cada uno (itemBuilder). Flutter llama a itemBuilder con index = 0, 1, 2...

NUEVO lib/presentation/home/productos_view.dart 
import 'package:flutter/material.dart';
import 'package:app_flutter/core/theme.dart';
import 'package:app_flutter/data/models/producto.dart';

// ======================================================
// Dibuja una lista de productos.
// NO sabe de donde vienen: los RECIBE por el constructor.
// Por eso podemos reutilizarla con cualquier lista.
// ======================================================

class ProductosView extends StatelessWidget {
  const ProductosView({super.key, required this.productos});

  final List<Producto> productos; // la lista que nos envian

  @override
  Widget build(BuildContext context) {
    // Si la lista esta vacia, mostramos un mensaje y terminamos aqui
    if (productos.isEmpty) {
      return const Center(child: Text('No hay productos para mostrar'));
    }

    return ListView.builder(
      padding: const EdgeInsets.all(8),
      itemCount: productos.length, // ¿cuantos elementos?
      // ¿como dibujo el elemento numero "index"?
      itemBuilder: (context, index) {
        final producto = productos[index]; // tomamos el producto de esa posicion

        return Card(
          child: ListTile(
            // leading = a la IZQUIERDA: circulo con la primera letra
            leading: CircleAvatar(
              backgroundColor: AppColors.secundario,
              foregroundColor: Colors.white,
              child: Text(producto.nombre[0]), // [0] = primera letra
            ),
            title: Text(producto.nombre),
            // toStringAsFixed(2) = siempre 2 decimales: 0.8 -> "0.80"
            subtitle: Text(
              '${producto.categoria} · S/ ${producto.precio.toStringAsFixed(2)}',
            ),
            // trailing = a la DERECHA: etiqueta con el stock
            trailing: Chip(
              label: Text('${producto.stock} und'),
              // Rojo si hay poco stock, verde si hay suficiente
              backgroundColor: producto.stockBajo
                  ? Colors.red.shade100
                  : Colors.green.shade100,
            ),
          ),
        );
      },
    );
  }
}

import 'package:flutter/material.dart';
import 'package:app_flutter/core/theme.dart';
import 'package:app_flutter/data/models/producto.dart';
import 'package:app_flutter/presentation/home/perfil_view.dart';

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key, required this.email});

  final String email; // sigue llegando desde el Login

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  // Pestaña seleccionada: 0 = Productos, 1 = Stock bajo, 2 = Perfil
  int currentScreenIndex = 0;

  // Copia de la lista de prueba. Los [...] crean una lista NUEVA
  // que si podemos modificar (la original es const).
  final List<Producto> productos = [...productosDemo];

  // Se ejecuta al presionar el boton +
  void agregarProducto() {
    setState(() {
      productos.add(
        Producto(
          nombre: 'Producto nuevo ${productos.length + 1}',
          categoria: 'Varios',
          precio: 1.00,
          stock: 1,
        ),
      );
    });
  }

  @override
  Widget build(BuildContext context) {
    // where() = FILTRAR. Nos quedamos solo con los de stock bajo.
    final productosBajos = productos.where((p) => p.stockBajo).toList();

    // Una vista por cada pestaña (mismo orden que el menu)
    final vistas = [
      ProductosView(productos: productos),       // 0
      ProductosView(productos: productosBajos),  // 1  (¡misma vista, otra lista!)
      PerfilView(email: widget.email),           // 2
    ];

    return Scaffold(
      appBar: AppBar(
        title: const Text('Inventario MiBodega'),
        backgroundColor: AppColors.primario,
        foregroundColor: Colors.white, // color del texto e iconos
      ),

      // Mostramos solo la vista de la pestaña elegida
      body: vistas[currentScreenIndex],

      // ---------- MENU INFERIOR ----------
      bottomNavigationBar: NavigationBar(
        selectedIndex: currentScreenIndex, // cual se ve "encendida"
        onDestinationSelected: (index) {
          // index = la pestaña que toco el usuario
          setState(() {
            currentScreenIndex = index;
          });
        },
        // Solo muestra el texto de la pestaña seleccionada
        labelBehavior: NavigationDestinationLabelBehavior.onlyShowSelected,
        destinations: [
          const NavigationDestination(
            icon: Icon(Icons.inventory_2_outlined),
            label: 'Productos',
          ),
          NavigationDestination(
            // Badge = globito con un numero encima del icono
            icon: Badge(
              label: Text('${productosBajos.length}'),
              child: const Icon(Icons.warning_amber_outlined),
            ),
            label: 'Stock bajo',
          ),
          const NavigationDestination(
            icon: Icon(Icons.person_outline),
            label: 'Perfil',
          ),
        ],
      ),

      // El boton + solo aparece en la pestaña Productos (index 0).
      // Si la condicion es falsa, ponemos null = no hay boton.
      floatingActionButton: currentScreenIndex == 0
          ? FloatingActionButton(
              onPressed: agregarProducto,
              child: const Icon(Icons.add),
            )
          : null,
    );
  }
}

// Tema general de la app. Lo usa MaterialApp en main.dart
// ======================================================
class AppTheme {
  static ThemeData tema() {
    return ThemeData(
      useMaterial3: true, // estilo moderno de Google (Material 3)

      // A partir de UN color, Flutter genera toda la paleta
      colorScheme: ColorScheme.fromSeed(seedColor: AppColors.primario),

      // Estilo para TODOS los AppBar de la app
      appBarTheme: const AppBarTheme(
        backgroundColor: AppColors.primario,
        foregroundColor: Colors.white,
        centerTitle: true, // titulo al centro
      ),

      // Estilo para TODAS las cajas de texto
      inputDecorationTheme: const InputDecorationTheme(
        border: OutlineInputBorder(),
        isDense: true,
      ),
    );
  }
}
