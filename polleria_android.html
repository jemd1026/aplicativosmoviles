1. Estructura del proyecto

package com.example.polleria.utils

import com.example.polleria.models.ItemCarrito
import com.example.polleria.models.Producto

object CarritoManager {
    val items = mutableListOf<ItemCarrito>()
    fun agregar(p: Producto): Boolean {
        val existente = items.find { it.producto.id == p.id }
        val cantidadNueva = (existente?.cantidad ?: 0) + 1
        if (p.stock < cantidadNueva) return false
        if (existente != null) existente.cantidad++ else items.add(ItemCarrito(p))
        return true
    }
    fun quitar(id: String) { items.removeAll { it.producto.id == id } }
    fun cambiar(id: String, incremento: Int) {
        val item = items.find { it.producto.id == id } ?: return
        val nuevo = item.cantidad + incremento
        if (nuevo <= 0) quitar(id) else if (nuevo <= item.producto.stock) item.cantidad = nuevo
    }
    fun total() = items.sumOf { it.subtotal() }
    fun limpiar() = items.clear()
}
Dato importante: el carrito solo vive mientras el proceso de la app esté activo. Si Android cierra la app, el carrito se pierde.

Kotlin · app/src/main/java/com/example/polleria/data/ProductoRepository.kt
package com.example.polleria.data

import com.example.polleria.models.Categoria
import com.example.polleria.models.Producto

object ProductoRepository {
    fun listarCategorias(onDone: (List<Categoria>, String?) -> Unit) {
        FirebaseManager.db.collection("categorias").whereEqualTo("activo", true).get()
            .addOnSuccessListener { docs ->
                onDone(docs.documents.mapNotNull { d -> d.toObject(Categoria::class.java)?.apply { id = d.id } }, null)
            }.addOnFailureListener { onDone(emptyList(), it.localizedMessage) }
    }
    fun listarProductos(categoria: String? = null, onDone: (List<Producto>, String?) -> Unit) {
        val q = FirebaseManager.db.collection("productos").whereEqualTo("activo", true)
        q.get().addOnSuccessListener { docs ->
            val productos = docs.documents.mapNotNull { d -> d.toObject(Producto::class.java)?.apply { id = d.id } }
            onDone(if (categoria == null) productos else productos.filter { it.idCategoria == categoria }, null)
        }.addOnFailureListener { onDone(emptyList(), it.localizedMessage) }
    }
}
Kotlin · app/src/main/java/com/example/polleria/data/PedidoRepository.kt
package com.example.polleria.data

import com.example.polleria.models.ItemCarrito
import com.google.firebase.Timestamp

object PedidoRepository {
    fun registrar(items: List<ItemCarrito>, onDone: (String?, String?) -> Unit) {
        val uid = FirebaseManager.auth.currentUser?.uid ?: run { onDone(null, "Sesión no iniciada"); return }
        if (items.isEmpty()) { onDone(null, "El carrito está vacío"); return }
        val db = FirebaseManager.db
        val pedido = db.collection("pedidos").document()
        val batch = db.batch()
        val total = items.sumOf { it.subtotal() }
        batch.set(pedido, mapOf(
            "uidCliente" to uid, "fechaPedido" to Timestamp.now(),
            "total" to total, "estado" to "Pendiente"
        ))
        items.forEach { item ->
            val doc = db.collection("detallePedidos").document()
            batch.set(doc, mapOf(
                "idPedido" to pedido.id, "idProducto" to item.producto.id,
                "nombreProducto" to item.producto.nombre, "cantidad" to item.cantidad,
                "precioUnitario" to item.producto.precio, "subtotal" to item.subtotal()
            ))
        }
        batch.commit().addOnSuccessListener { onDone(pedido.id, null) }
            .addOnFailureListener { onDone(null, it.localizedMessage) }
    }
}
Escritura por lotes: PedidoRepository usa db.batch() para guardar el pedido y sus detalles juntos. Si falla un documento, ninguno se guarda.

Reglas de seguridad de Firestore (práctica)

Reglas Firestore · firestore.rules.EJEMPLO
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function autenticado() { return request.auth != null; }
    match /categorias/{id} { allow read: if autenticado(); allow write: if false; }
    match /productos/{id} { allow read: if autenticado(); allow write: if false; }
    match /clientes/{uid} {
      allow read: if autenticado() && request.auth.uid == uid;
      allow create: if autenticado() && request.auth.uid == uid && request.resource.data.uid == uid;
      allow update, delete: if false;
    }
    // Solo ejemplo de clase, NO suficiente para una tienda real:
    // El cliente podría adulterar precios, total y contenido.
    match /pedidos/{id} {
      allow create: if autenticado() && request.resource.data.uidCliente == request.auth.uid
                    && request.resource.data.estado == 'Pendiente';
      allow read: if autenticado() && resource.data.uidCliente == request.auth.uid;
      allow update, delete: if false;
    }
    match /detallePedidos/{id} {
      // Esquema académico: la verificación cruzada entre detalles y pedido
      // requiere reglas más complejas o lógica en un backend de confianza.
      allow create: if autenticado()
        && request.resource.data.idPedido is string
        && getAfter(/databases/$(database)/documents/pedidos/$(request.resource.data.idPedido)).data.uidCliente == request.auth.uid;
      allow read: if false;
      allow update, delete: if false;
    }
  }
}
Advertencia: estas reglas son solo para clase. No las uses en producción sin revisarlas.

package com.example.polleria.adapters

import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.TextView
import androidx.recyclerview.widget.RecyclerView
import com.example.polleria.R
import com.example.polleria.models.Producto

class ProductoAdapter(private val onClick: (Producto) -> Unit) : RecyclerView.Adapter<ProductoAdapter.Holder>() {
    private val datos = mutableListOf<Producto>()
    fun actualizar(lista: List<Producto>) { datos.clear(); datos.addAll(lista); notifyDataSetChanged() }
    class Holder(v: View) : RecyclerView.ViewHolder(v) {
        val nombre: TextView = v.findViewById(R.id.tvNombreProducto)
        val descripcion: TextView = v.findViewById(R.id.tvDescripcionProducto)
        val precio: TextView = v.findViewById(R.id.tvPrecioProducto)
    }
    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int) = Holder(
        LayoutInflater.from(parent.context).inflate(R.layout.item_producto,parent,false))
    override fun getItemCount() = datos.size
    override fun onBindViewHolder(holder: Holder, position: Int) {
        val p = datos[position]
        holder.nombre.text = p.nombre
        holder.descripcion.text = p.descripcion
        holder.precio.text = "S/ %.2f".format(p.precio)
        holder.itemView.setOnClickListener { onClick(p) }
    }
}

import android.view.LayoutInflater

import android.widget.TextView
import androidx.recyclerview.widget.RecyclerView
import com.example.polleria.R
import com.example.polleria.models.ItemCarrito
import com.example.polleria.utils.CarritoManager

class CarritoAdapter(private val onChange: () -> Unit): RecyclerView.Adapter<CarritoAdapter.Holder>() {
    private var datos: List<ItemCarrito> = emptyList()
    fun actualizar() { datos = CarritoManager.items.toList(); notifyDataSetChanged() }
    class Holder(v: View): RecyclerView.ViewHolder(v) {
        val texto: TextView = v.findViewById(R.id.tvItemCarrito)
        val menos: Button = v.findViewById(R.id.btnMenos)
        val mas: Button = v.findViewById(R.id.btnMas)
        val quitar: Button = v.findViewById(R.id.btnQuitar)
    }
    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int) = Holder(LayoutInflater.from(parent.context).inflate(R.layout.item_carrito,parent,false))
    override fun getItemCount() = datos.size
    override fun onBindViewHolder(holder: Holder, position: Int) {
        val i = datos[position]
        holder.texto.text = "${i.producto.nombre} · ${i.cantidad} und.\nS/ %.2f".format(i.subtotal())
        holder.menos.setOnClickListener { CarritoManager.cambiar(i.producto.id,-1); onChange() }
        holder.mas.setOnClickListener { CarritoManager.cambiar(i.producto.id,1); onChange() }
        holder.quitar.setOnClickListener { CarritoManager.quitar(i.producto.id); onChange() }
    }
}

class MainActivity : AppCompatActivity() {

        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        val barra = findViewById<BottomNavigationView>(R.id.bottomNav)
        barra.setOnItemSelectedListener { item ->
            val f: Fragment = when (item.itemId) {
                R.id.nav_categorias -> CategoriasFragment()
                R.id.nav_carrito -> CarritoFragment()
                R.id.nav_perfil -> PerfilFragment()
                else -> CatalogoFragment()
            }
            supportFragmentManager.beginTransaction().replace(R.id.fragmentContainer, f).commit()
            true
        }
        if (savedInstanceState == null) barra.selectedItemId = R.id.nav_catalogo
    }
}

LoginActivity (inicio de sesión)

Kotlin · app/src/main/java/com/example/polleria/activities/LoginActivity.kt
package com.example.polleria.activities

import android.content.Intent
import android.os.Bundle
import android.widget.Button
import android.widget.EditText
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import com.example.polleria.R
import com.example.polleria.data.FirebaseManager

class LoginActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        if (FirebaseManager.auth.currentUser != null) { abrirPrincipal(); return }
        setContentView(R.layout.activity_login)
        val correo = findViewById<EditText>(R.id.etCorreo)
        val clave = findViewById<EditText>(R.id.etClave)
        val entrar = findViewById<Button>(R.id.btnIngresar)
        entrar.setOnClickListener {
            val email = correo.text.toString().trim()
            val pass = clave.text.toString()
            if (!android.util.Patterns.EMAIL_ADDRESS.matcher(email).matches() || pass.isBlank()) {
                Toast.makeText(this, "Ingrese correo válido y contraseña", Toast.LENGTH_SHORT).show(); return@setOnClickListener
            }
            entrar.isEnabled = false
            FirebaseManager.auth.signInWithEmailAndPassword(email, pass)
                .addOnSuccessListener { abrirPrincipal() }
                .addOnFailureListener { Toast.makeText(this, "No se pudo iniciar sesión: ${it.localizedMessage}", Toast.LENGTH_LONG).show(); entrar.isEnabled = true }
        }
        findViewById<Button>(R.id.btnIrRegistro).setOnClickListener { startActivity(Intent(this, RegistroActivity::class.java)) }
    }
    private fun abrirPrincipal() {
        startActivity(Intent(this, MainActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK))
        finish()
    }
}

package com.example.polleria.activities

import android.content.Intent
import android.os.Bundle
import android.widget.Button
import android.widget.EditText
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import com.example.polleria.R
import com.example.polleria.data.FirebaseManager
import com.google.firebase.Timestamp

class RegistroActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_registro)
        val nombres = findViewById<EditText>(R.id.etNombres)
        val apellidos = findViewById<EditText>(R.id.etApellidos)
        val dni = findViewById<EditText>(R.id.etDni)
        val celular = findViewById<EditText>(R.id.etCelular)
        val direccion = findViewById<EditText>(R.id.etDireccion)
        val correo = findViewById<EditText>(R.id.etCorreoRegistro)
        val clave = findViewById<EditText>(R.id.etClaveRegistro)
        val guardar = findViewById<Button>(R.id.btnRegistrar)
        guardar.setOnClickListener {
            val email = correo.text.toString().trim()
            val password = clave.text.toString()
            if (listOf(nombres,apellidos,celular,direccion).any { it.text.toString().isBlank() } ||
                !dni.text.toString().matches(Regex("[0-9]{8}")) ||
                !android.util.Patterns.EMAIL_ADDRESS.matcher(email).matches() || password.length < 6) {
                Toast.makeText(this, "Complete los datos. DNI: 8 dígitos y contraseña: mínimo 6 caracteres", Toast.LENGTH_LONG).show()
                return@setOnClickListener
            }
            guardar.isEnabled = false
            FirebaseManager.auth.createUserWithEmailAndPassword(email,password)
                .addOnSuccessListener { result ->
                    val uid = result.user?.uid ?: return@addOnSuccessListener
                    val cliente = mapOf(
                        "uid" to uid, "nombres" to nombres.text.toString().trim(),
                        "apellidos" to apellidos.text.toString().trim(), "dni" to dni.text.toString(),
                        "celular" to celular.text.toString(), "direccion" to direccion.text.toString().trim(),
                        "correo" to email, "fechaRegistro" to Timestamp.now(), "activo" to true
                    )
                    FirebaseManager.db.collection("clientes").document(uid).set(cliente)
                        .addOnSuccessListener {
                            startActivity(Intent(this, MainActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_CLEAR_TASK or Intent.FLAG_ACTIVITY_NEW_TASK))
                            finish()
                        }
                        .addOnFailureListener { e ->
                            Toast.makeText(this, "Cuenta creada, pero no se guardó el perfil: ${e.localizedMessage}. Reintente el registro del perfil con ayuda del docente.", Toast.LENGTH_LONG).show()
                            guardar.isEnabled = true
                        }
                }.addOnFailureListener {
                    Toast.makeText(this, it.localizedMessage ?: "Error de registro", Toast.LENGTH_LONG).show()
                    guardar.isEnabled = true
                }
        }
    }
}

# Pollería — Android (Kotlin + XML) + Firebase

Paquete: `com.example.polleria`.

## 1. Copiar archivos

Conserva el proyecto **Polleria** que acabas de crear en Android Studio. Copia el contenido de esta carpeta `app/src/main/` sobre el `app/src/main/` del proyecto existente. El archivo `MainActivity.kt` antiguo de la raíz `com.example.polleria` debe eliminarse porque ahora está en `com.example.polleria.activities`. No debes conservar dos actividades launcher en el AndroidManifest. Los otros recursos que Android Studio creó (íconos, `themes.xml`, etc.) pueden permanecer.

**Importante:** es un paquete de código fuente, no un proyecto Android Studio autónomo, porque tus versiones exactas de Gradle, AGP y Kotlin no aparecen en la captura.

## 2. Firebase (paso obligatorio)

1. En https://console.firebase.google.com, crea un proyecto.
2. Agrega una app Android con el ID exacto `com.example.polleria`.
3. Descarga `google-services.json` y colócalo en `<tu-proyecto>/app/google-services.json` (archivo real de TU Firebase; no se puede generar genéricamente).
4. Activa **Authentication > Sign-in method > Email/Password**.
5. Activa **Cloud Firestore** y configura las reglas. Para clase, ver el archivo `firestore.rules.EJEMPLO`; **no uses reglas abiertas en producción**.

## 3. Gradle Kotlin DSL (build.gradle.kts)

Añade al bloque `plugins` del **proyecto raíz** el plugin de Google Services con una versión compatible con tu proyecto (ejemplo: `id("com.google.gms.google-services") version "4.4.2" apply false`).

En `app/build.gradle.kts`, añade al bloque `plugins`:

```kotlin
id("com.google.gms.google-services")
```

En el bloque `dependencies` del módulo `app` (conserva las dependencias existentes):

```kotlin
implementation(platform("com.google.firebase:firebase-bom:34.4.0"))
implementation("com.google.firebase:firebase-auth")
implementation("com.google.firebase:firebase-firestore")
implementation("androidx.recyclerview:recyclerview:1.4.0")
implementation("com.google.android.material:material:1.12.0")
implementation("androidx.appcompat:appcompat:1.7.1")
implementation("androidx.fragment:fragment-ktx:1.8.9")
```

**Versiones de referencia:** comprueba la compatibilidad con la versión instalada de Android Studio/AGP y actualiza versiones según el IDE. RecyclerView 1.4.0 requiere compilar con API 35 o superior. Si tienes un `build.gradle` Groovy, convierte la sintaxis.

Sincroniza Gradle (`Sync Now`) y ejecuta en el emulador con Internet.

Se crean **manualmente** desde Firebase Console estas colecciones:

`categorias`:
- documento ID: `pollos`: `nombre` (string) = `Pollos a la brasa`, `descripcion` = `Clásicos`, `activo` (boolean) = `true`
- documento ID: `combos`: `nombre` = `Combos`, `descripcion` = `Combos familiares`, `activo` = `true`
- documento ID: `bebidas`: `nombre` = `Bebidas`, `descripcion` = `Refrescos`, `activo` = `true`

`productos`:
- documento ID: `pollo_entero`: `idCategoria` = `pollos`, `nombre` = `Pollo entero`, `descripcion` = `Con papas y ensalada`, `precio` (number) = `65.0`, `imagenUrl` = ``, `stock` (number) = `20`, `activo` = `true`
- documento ID: `medio_pollo`: `idCategoria` = `pollos`, `nombre` = `Medio pollo`, `descripcion` = `Con papas`, `precio` = `36.0`, `imagenUrl` = ``, `stock` = `20`, `activo` = `true`
- documento ID: `combo_familiar`: `idCategoria` = `combos`, `nombre` = `Combo familiar`, `descripcion` = `Pollo, papas y gaseosa`, `precio` = `79.0`, `imagenUrl` = ``, `stock` = `10`, `activo` = `true`
- documento ID: `chicha`: `idCategoria` = `bebidas`, `nombre` = `Chicha morada`, `descripcion` = `1 litro`, `precio` = `10.0`, `imagenUrl` = ``, `stock` = `30`, `activo` = `true`

Todos los nombres de campo distinguen mayúsculas y minúsculas; `precio` y `stock` deben tener **tipo número**, `activo` debe ser **booleano**.

La app escribe en `clientes`, `pedidos` y `detallePedidos` cuando se registra y confirma un pedido.

## 5. Recorrido para probar

1. Crear cuenta con correo/contraseña y datos del cliente.
2. Ver catálogo; tocar un producto; pulsar Agregar al carrito.
3. Abrir menú Carrito; cambiar cantidad; continuar pedido.
4. Confirmar; verificar en Firestore el documento `pedidos` y documentos relacionados en `detallePedidos`.
5. Abrir Cuenta y cerrar sesión; ingresar otra vez.

## 6. Temas de aprendizaje

- Activity: `LoginActivity`, `RegistroActivity`, `DetalleProductoActivity`, `ConfirmarPedidoActivity`.
- Fragment y ciclo de vida: `CatalogoFragment`, `CategoriasFragment`, `CarritoFragment`, `PerfilFragment`.
- RecyclerView + ViewHolder + Adapter: `ProductoAdapter`, `CategoriaAdapter`, `CarritoAdapter`.
- POO/Data class: `Producto`, `Categoria`, `ItemCarrito`.
- Firebase Auth: `FirebaseManager`, `LoginActivity`, `RegistroActivity`.
- Firestore y escrituras por lotes: `ProductoRepository`, `PedidoRepository`.
- Datos locales temporales: `CarritoManager` (se pierde si Android cierra el proceso).

## 7. Límites importantes de esta versión académica

- **No es una tienda lista para producción**: la validación del total, precios, stock, límites y autorización para escribir pedidos debería hacerse en un servidor o Cloud Functions con Firebase Admin SDK, no confiando en campos enviados desde el cliente.
- El stock se verifica en la pantalla para práctica, pero NO se descuenta de Firestore ni es transaccional.
- Las imágenes se omiten deliberadamente del RecyclerView, aunque `imagenUrl` está incluido en el modelo. Se pueden agregar con Coil/Glide en una segunda etapa.
- El carrito está en memoria: no persiste entre reinicios del proceso.
- La creación de la cuenta en Authentication y escritura del perfil no es una transacción atómica; si falla la escritura del perfil, el docente debe revisar y corregir el estado en Firebase.
- Los listeners `get()` leen una vez; no es catálogo realtime. No se incluye consulta de historial ni verificación de correo.
- Este entregable no ha sido compilado contra tu instalación específica de Android Studio; la configuración Gradle y las reglas de tu Firebase deben verificarse localmente.

PARTE VIII: LÍMITES DE ESTA VERSIÓN

No es una tienda lista para producción. Esta versión es académica y tiene estas limitaciones:
  • El total, los precios y el stock se calculan en la app. En un sistema real esto debe validarse en un servidor o en Cloud Functions.
  • El stock se verifica en pantalla, pero no se descuenta de Firestore ni es transaccional.
  • El carrito está en memoria: no persiste entre reinicios del proceso.
  • Las imágenes (imagenUrl) están en el modelo, pero no se muestran en el RecyclerView. Se pueden agregar con Coil o Glide.
  • La cuenta en Authentication y el perfil en Firestore no son atómicos: si falla el perfil, hay que corregirlo a mano.
  • Las consultas usan get(), que lee una sola vez; no es un catálogo en tiempo real.
